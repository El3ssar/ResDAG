    Hyperparameter tuning¶

    model.tune(data) is the common entry point for rd.ESN and rd.Model. It compares autonomous forecasts, installs a refitted winner, and returns the same outer PyTorch module. The standalone resdag.hpo.tune_model(model, data, ...) accepts the same advanced options.

    Install resdag[hpo] for Optuna and joblib. Add the dynamics extra to accept TSDynamics trajectories on Python 3.12 or newer.

    Common tuning API¶

    tuning ¶

    Common, transactional tuning of fitted PyTorch reservoir models.

    Optuna owns sampling and trial bookkeeping; joblib owns parallel processes. The data adapter owns immutable chronological views. Candidates use the same fit/forecast workflow as the final model, including named ports and units.

    ScoreContext dataclass ¶

    ScoreContext(
    predictions: dict[str, Tensor],
    truth: dict[str, Tensor],
    scales: dict[str, Tensor],
    seed: int,
    origin: int,
)

    Original-unit tensors and fixed training scales for a custom objective.

    Return one finite scalar per seed/origin. The trial score is their median. Set direction='minimize' when returning an error rather than a reward.

    Explicitly prefix-safe pruning objective with the complete target horizon.

    Predictions and truth contain only the admitted prefix. This callback controls pruning, while objective still scores the complete forecast. Keep its direction and scale comparable across matching resource steps.

    def tune_model(
    model: Model | ESN,
    training: Any,
    targets: Any = None,
    *,
    preset: Preset = "balanced",
    search_space: SearchSpace | None = None,
    factory: Factory | None = None,
    validation: tuple[Any, Any] | None = None,
    validation_fraction: float = 0.2,
    horizon: int | None = None,
    n_trials: int | None = None,
    seeds: Sequence[int] | None = None,
    seed: int = 0,
    n_origins: int | None = None,
    threshold: float = 0.2,
    aggregation: Literal["weakest", "mean"] = "weakest",
    objective: Callable[[ScoreContext], float] | None = None,
    pruning_objective: Callable[[PrefixScoreContext], float] | None = None,
    direction: Literal["maximize", "minimize"] = "maximize",
    sampler: optuna.samplers.BaseSampler | None = None,
    pruner: str | optuna.pruners.BasePruner | None = None,
    checkpoint_interval: int = 64,
    study: optuna.Study | None = None,
    storage: str | None = None,
    study_name: str | None = None,
    warm_start: optuna.Study | None = None,
    transfer_mode: Literal["reevaluate", "scores"] = "reevaluate",
    screening: bool | Screening | None = False,
    memory_budget: MemoryBudget | str | int | None = None,
    memory_estimator: Callable[[dict[str, Any], MemoryContext], MemoryEstimate | int] | None = None,
    n_jobs: int = 1,
    timeout: float | None = None,
    inner_max_num_threads: int = 1,
    catch_exceptions: bool = True,
    execution: Literal["auto", "fast", "history"] = "auto",
) -> Model | ESN:
    """Compare autonomous forecasts and install a fully refitted winner.

    `factory(params, seed)` builds custom candidate DAGs. Without a factory,
    custom models keep their weights/wiring and tune alpha/washout; presets
    rebuild automatically. New attempts, including failed configurations, count
    toward `n_trials`. A timeout stops new windows; running fits and final
    refitting finish normally. Tune before constructing a PyTorch optimizer.
    """
    started = time.perf_counter()
    if preset not in _PRESETS:
        raise ValueError("preset must be 'quick', 'balanced' or 'thorough'")
    if factory is not None and isinstance(model, ESN):
        raise ValueError("ESN rebuilds itself automatically; use rd.Model for a custom factory")
    if aggregation not in ("weakest", "mean"):
        raise ValueError("aggregation must be 'weakest' or 'mean'")
    if execution not in ("auto", "fast", "history"):
        raise ValueError("execution must be 'auto', 'fast' or 'history'")
    if direction not in ("maximize", "minimize"):
        raise ValueError("direction must be 'maximize' or 'minimize'")
    if objective is None and direction != "maximize":
        raise ValueError("The built-in horizon objective requires direction='maximize'")
    if not math.isfinite(threshold) or threshold <= 0:
        raise ValueError("threshold must be positive and finite")
    if timeout is not None and (not math.isfinite(timeout) or timeout <= 0):
        raise ValueError("timeout must be positive and finite")
    if isinstance(n_jobs, bool) or not isinstance(n_jobs, int) or n_jobs == 0:
        raise ValueError("n_jobs must be a nonzero integer")
    _count(inner_max_num_threads, "inner_max_num_threads")
    _count(checkpoint_interval, "checkpoint_interval")
    if not isinstance(catch_exceptions, bool):
        raise TypeError("catch_exceptions must be a boolean")
    if isinstance(seed, bool) or not isinstance(seed, int) or not 0 <= seed < 2**63:
        raise ValueError("seed must be an integer in [0, 2**63)")
    trials, seed_count, origin_count = _PRESETS[preset]
    trials = trials if n_trials is None else _count(n_trials, "n_trials")
    origin_count = origin_count if n_origins is None else _count(n_origins, "n_origins")
    fixed_graph = factory is None and _automatic_options(model) is None
    if seeds is None:
        seeds = tuple(seed + i for i in range(1 if fixed_graph else seed_count))
    try:
        seeds = tuple(seeds)
    except TypeError as error:
        raise ValueError("seeds must be a nonempty sequence of integers") from error
    if not seeds or any(
        isinstance(s, bool) or not isinstance(s, int) or not 0 <= s < 2**63 for s in seeds
    ):
        raise ValueError("seeds must contain integers in [0, 2**63)")
    seed_bank = tuple(dict.fromkeys(seeds))
    if fixed_graph and len(seed_bank) != 1:
        raise ValueError("Several initialization seeds require a factory to rebuild a custom DAG")
    if study is not None and (
        storage is not None or study_name is not None or sampler is not None or pruner is not None
    ):
        raise ValueError("Pass an existing study or storage/study_name/sampler/pruner, not both")
    if warm_start is not None and not isinstance(warm_start, optuna.Study):
        raise TypeError("warm_start must be an Optuna study")
    if warm_start is not None and warm_start is study:
        raise ValueError("warm_start must be a separate source study")
    if transfer_mode not in ("reevaluate", "scores"):
        raise ValueError("transfer_mode must be 'reevaluate' or 'scores'")
    if search_space is not None and not callable(search_space):
        raise TypeError("search_space must be callable")
    if factory is not None and not callable(factory):
        raise TypeError("factory must be callable")
    if objective is not None and not callable(objective):
        raise TypeError("objective must be callable")
    if pruning_objective is not None and not callable(pruning_objective):
        raise TypeError("pruning_objective must be callable")
    if screening is True:
        screening = Screening()
    elif screening is False:
        screening = None
    elif screening is not None and not isinstance(screening, Screening):
        raise TypeError("screening must be False, True, or a Screening policy")
    budget = (
        memory_budget
        if isinstance(memory_budget, MemoryBudget) or memory_budget is None
        else MemoryBudget(memory_budget)
    )
    if memory_estimator is not None and not callable(memory_estimator):
        raise TypeError("memory_estimator must be callable")
    if memory_estimator is not None and budget is None:
        raise ValueError("memory_estimator requires memory_budget")
    if budget is not None and factory is not None and memory_estimator is None:
        raise ValueError("A custom factory with a memory budget requires memory_estimator")
    resolved_pruner: optuna.pruners.BasePruner
    if study is None and isinstance(pruner, str) and pruner.lower() == "threshold":
        if objective is not None:
            raise ValueError(
                "The common threshold preset is for positive horizon scores; "
                "supply a configured Optuna ThresholdPruner for a custom objective"
            )
        resolved_pruner = optuna.pruners.ThresholdPruner(
            lower=0.1, n_warmup_steps=2 * checkpoint_interval
        )
    else:
        resolved_pruner = resolve_pruner(pruner) if study is None else study.pruner
    # A huge lazy preset must not allocate its initial dense recurrence before
    # resource planning can even start. This is a known lower bound, not a
    # substituted smaller architecture or a claim about solver peak memory.
    if budget is not None and isinstance(model, ESN) and model.workflow is None:
        size = model._factory_options["reservoir_size"]
        sample = training.get("input") if isinstance(training, Mapping) else training
        if is_trajectory(sample):
            sample = torch.as_tensor(cast(Any, sample).y)
        element = model._reference.element_size()
        if not model._dtype_pinned and isinstance(sample, torch.Tensor):
            element = sample.element_size()
        lower = size * size * element
        destination = (
            model._reference.device
            if model._device_pinned or not isinstance(sample, torch.Tensor)
            else sample.device
        )
        minimum = MemoryEstimate(lower) if destination.type == "cpu" else MemoryEstimate(0, lower)
        if not budget.fits(minimum):
            raise ValueError(
                "The lazy ESN base template exceeds memory_budget before construction; "
                "start from a smaller base model or provide a custom construction recipe"
            )
    previous_report = model.__dict__.get("tuning_report")
    resuming = isinstance(previous_report, TuningReport) and (
        study is previous_report.study
        or (storage is not None and study_name == previous_report.study.study_name)
    )
    if resuming:
        assert isinstance(previous_report, TuningReport)
        if _model_hash(model) != previous_report._installed_hash:
            raise ValueError(
                "The model changed since tuning; use a fresh study for a changed model"
            )
        template = _template(previous_report._template, training, seed)
    else:
        template = _template(model, training, seed)
    workflow = _workflow(template)
    data = prepare_tuning_data(
        workflow,
        training,
        targets,
        validation=validation,
        validation_fraction=validation_fraction,
        horizon=horizon,
        n_origins=origin_count,
    )
    if screening is not None:
        screening.validate(len(seed_bank), len(data.origins))
    memory = (
        None
        if budget is None
        else _memory_context(
            workflow,
            data,
            preset_options=_automatic_options(template),
        )
    )
    if memory is not None and (objective is not None or pruning_objective is not None):
        reference = module_reference(workflow.graph)
        element = 4 if reference is None else reference.element_size()
        batch = next(iter(data.origins[0].truth.values()))
        batches = batch.shape[0] if batch.ndim == 3 else 1
        retained = (
            batches
            * max(origin.horizon for origin in data.origins)
            * sum(workflow._output_widths)
            * element
        )
        memory = replace(memory, retained_forecast_bytes=retained)
    if memory is not None and (
        execution == "history" or (execution == "auto" and workflow._requires_complete_history())
    ):
        element = torch.finfo(memory.dtype).bits // 8
        replay = max(
            sum(
                (value.shape[0] if value.ndim == 3 else 1)
                * (value.shape[-2] + origin.horizon)
                * value.shape[-1]
                for value in origin.history.values()
            )
            for origin in data.origins
        )
        memory = replace(
            memory,
            replay_history_bytes=3 * element * replay,
            forecast_samples=max(
                (value.shape[0] if value.ndim == 3 else 1) * (value.shape[-2] + origin.horizon)
                for origin in data.origins
                for value in origin.history.values()
            ),
        )
    automatic = _automatic_options(template)
    recipe = factory or (
        _preset_recipe(template) if automatic is not None else _fixed_recipe(template)
    )
    space = search_space or _default_space(template)
    fingerprint, _ = _protocol(
        data,
        template,
        seeds=seed_bank,
        threshold=threshold,
        aggregation=aggregation,
        direction=direction,
        factory=_callable_identity(factory),
        search_space=_callable_identity(search_space),
        objective=_callable_identity(objective),
        pruning_objective=_callable_identity(pruning_objective),
        memory_budget=None if budget is None else asdict(budget),
        memory_estimator=_callable_identity(memory_estimator),
        fixed_graph=fixed_graph,
        builtin_recipe=(
            None
            if automatic is None
            else {
                "version": 1,
                "family": cast(dict[str, Any], builtin_recipe(template.graph))["family"],
                "options": _configuration(automatic),
                "default_space_version": 2 if search_space is None else None,
            }
        ),
        execution=execution,
    )
    owned = None
    primary_error: BaseException | None = None
    try:
        if study is None:
            owned = resolve_storage(storage, n_workers=1)
            study = optuna.create_study(
                storage=owned,
                study_name=study_name,
                load_if_exists=study_name is not None,
                direction=direction,
                sampler=sampler or optuna.samplers.TPESampler(seed=seed),
                pruner=resolved_pruner,
            )
        if len(study.directions) != 1 or study.direction.name.lower() != direction:
            raise ValueError("Study direction must match the tuning objective")
        bind_protocol(study, fingerprint)
        if warm_start is not None:
            from .transfer import transfer_trials

            transfer_trials(study, warm_start, mode=transfer_mode, protocol_hash=fingerprint)
        study.set_user_attr(
            "resdag_tuning",
            {
                "version": 2,
                "seeds": list(seed_bank),
                "refit_seed": seed_bank[0],
                "validation": data.report,
                "threshold": threshold,
                "aggregation": aggregation,
                "direction": direction,
                "fixed_graph": fixed_graph,
            },
        )
        reference = module_reference(workflow.graph)
        evaluator = _Evaluator(
            recipe,
            data,
            seed_bank,
            threshold,
            aggregation,
            objective,
            catch_exceptions,
            torch.device("cpu") if reference is None else reference.device,
            execution,
            checkpoint_interval,
            pruning_objective,
        )
        attempts, resource_report = _run_trials(
            study,
            evaluator,
            space,
            trials=trials,
            seed=seed,
            n_jobs=n_jobs,
            inner_max_num_threads=inner_max_num_threads,
            started=started,
            timeout=timeout,
            direction=direction,
            screening=screening,
            budget=budget,
            context=memory,
            custom_factory=factory is not None,
            estimator=memory_estimator,
        )
        study.set_user_attr("resdag_tuning_resources", resource_report)
        valid = [
            trial
            for trial in study.trials
            if trial.state == optuna.trial.TrialState.COMPLETE
            and trial.user_attrs.get("resdag_tune_valid")
            and trial.value is not None
            and math.isfinite(trial.value)
        ]
        if not valid:
            detail = next(
                (
                    trial.user_attrs.get("error", trial.user_attrs.get("pruning_reason"))
                    for trial in study.trials
                    if "error" in trial.user_attrs or "pruning_reason" in trial.user_attrs
                ),
                "no evaluation finished",
            )
            raise RuntimeError(f"No valid tuning candidates completed: {detail}")
        best = (max if direction == "maximize" else min)(
            valid, key=lambda trial: cast(float, trial.value)
        )
        if study.best_trial.number != best.number:
            raise ValueError(
                "Study contains a winning score outside this tuning protocol; use a fresh study"
            )
        best_params = ast.literal_eval(best.user_attrs["resolved_params_literal"])
        refit_started = time.perf_counter()
        with _isolated_seed(seed_bank[0], evaluator.device):
            winner = evaluator.build(best_params, seed_bank[0])
            data.fit(_workflow(winner), refit=True)
            _synchronize(_workflow(winner))
        refit_seconds = time.perf_counter() - refit_started
        report = TuningReport(
            study,
            best_params,
            best.number,
            cast(float, best.value),
            best.user_attrs["nrmse"],
            seed_bank,
            seed_bank[0],
            data.report,
            fingerprint,
            attempts,
            time.perf_counter() - started,
            refit_seconds,
            tuple(best.user_attrs["runs"]),
            template,
            _model_hash(winner),
            resource_report,
        )
        _close_storage(owned, keep_study_usable=True)
        _install(model, winner, report)
        owned = None
        return model
    except BaseException as error:
        primary_error = error
        raise
    finally:
        _close_storage(owned, primary_error=primary_error, keep_study_usable=primary_error is None)

    Data and validation¶

    Argument Default Contract
    training required Floating (time, features) or (batch, time, features) tensors, a named tensor mapping, a bare trajectory, or named trajectory inputs.
    targets None Explicit named output targets, or implicit future observations for one input/output of equal width. Tensor targets are already aligned to input origins; trajectory targets are sampled at each head's physical prediction lead.
    validation None (inputs, targets_or_None) for an explicit later segment. It is excluded from both candidate fitting and final refitting.
    validation_fraction 0.2 Chronological held-out fraction when validation is automatic.
    horizon None Complete scored length per origin. Default: half the available validation span, at least one sample. It stays the same across presets and candidates.
    n_origins preset Number of distinct, evenly spread supported forecast starts; reduced if fewer complete windows exist.

    Automatic validation fits candidates on the earlier observations, then refits the winning configuration on all supplied development observations. Keep a separate final test set. Explicit validation stays held out during refitting; supply its later observations as forecast history when continuing after it.

    Every origin uses the candidate's fixed readout and preprocessing. A later origin can synchronize on earlier validation observations without fitting on them. Complete windows can overlap. No unsupported horizon is silently shortened.

    For an observed Tensor fit window of length C and largest head lead L, shared supervised origins stop before C - L. This prevents future labels crossing the fit boundary. The complete observed inputs are still retained for forecast history. The same physical-support rule applies during final refitting; validation["refit_discarded_target_rows"] reports discarded rows by head. Standalone fit(inputs, tensor_targets) trusts the caller's aligned rows; tuning deliberately caps them to the observed development window.

    With explicit origin-aligned Tensor validation, the first scored forecast starts after max(prediction_leads) validation observations. For example, leads one and three skip the first three validation observations before selecting scored origins. first_validation_index includes the preceding training length. Trajectory truth is queried at physical dates and uses its actual source support instead. Inputs and validation targets must preserve component names/order.

    Trajectory interpolation is prepared once per fixed view from raw source slices. Training and origin histories cannot use later source values in interpolation coefficients. A named clock can require unscored preroll to reach the requested validation date; reports preserve its actual endpoint, first scored date and skip count. Forecasts still receive only declared non-feedback future drivers.

    Search and repetitions¶

    Argument Default Contract
    preset "balanced" quick: 8 trials/1 seed/1 origin; balanced: 24/3/3; thorough: 80/5/5.
    n_trials preset Positive budget of new attempts in this call; failures, pruning, memory rejection and screening rejection count. Timeout can stop early.
    search_space built-in search_space(optuna_trial) -> mapping; ordinary Optuna suggestions and conditional branches. Return fixed settings alongside sampled settings.
    factory built-in factory(params, seed) -> Model or ESN, rebuilding custom candidates. Accepted on a Model; the ESN preset rebuilds itself.
    seed 0 Sampler seed and start of the default initialization seed bank.
    seeds preset Explicit nonempty seed sequence; duplicates are removed. The same bank evaluates every configuration.

    Built-in rd.ESN and unmodified classic/OTT graphs wrapped in rd.Model use the following space. All presets use the same ranges; presets change the number of trials, seeds and starting points.

    Setting Default range Sampling
    reservoir_size Half to twice the initial width, each bound at least eight Integer
    spectral_radius 0.05–1.5 Uniform
    leak_rate 0.05–1.0 Uniform
    alpha 1e-12–1e-2 Logarithmic, when the readout exposes this setting
    input_scaling 1e-3–2.0 Logarithmic, when supported by the input initializer
    bias_scaling 1e-3–1.0 Logarithmic, when bias uses its default initializer
    mean_degree Three to twelve, capped by available recurrent inputs Integer, for the default or configured Erdős–Rényi topology
    input_to_readout True, False Categorical, vanilla recipes only; input plus states or states alone

    mean_degree is a tuning control for the automatic recipe. It controls the expected number of recurrent matrix entries per unit, including one for a permitted self-connection. It determines the Erdős–Rényi probability for the candidate's sampled width; the actual sampled connection count varies. Existing topology flags and initializer settings are preserved. A custom non-Erdős–Rényi topology, disabled bias or custom bias initializer keeps its own construction rather than being replaced merely to expose another search dimension.

    Named input initializers and InputFeedbackInitializer objects support the shared scaling control without losing their other options. Arbitrary matrix callables and FunctionInitializer objects keep their original construction; the tuner cannot assume that they accept or honor a scaling keyword. Supply an explicit factory when you want to search their own settings.

    search_space replaces the default space completely. Return only the settings you want to vary or fix, using ordinary Optuna suggestions for any ranges. Current workflow settings and custom transforms are copied into candidates; the winner is refitted rather than inheriting fitted transform statistics.

    Without a factory, a custom Model keeps its DAG and reservoir weights. This also applies when the structure or reservoir weights of a built-in graph have been edited after construction: automatic rebuilding must not discard edits. The default search varies alpha; supplied settings may contain only alpha and washout. It evaluates one initialization seed, since copying frozen weights does not create a different reservoir. Its readouts must support the chosen setting. A factory enables structural changes and multiple initialization seeds.

    Ordinary fitting, recurrent memory, device/dtype changes, copying and portable save/load retain an eligible built-in recipe. A portable graph whose initializer recipe contains unsupported local Python code still saves its materialized weights; its automatic rebuilding recipe is omitted. After loading, use a custom factory to tune its architecture, or a trusted bundle when you need to retain that local Python recipe. See save and load.

    Recipes can vary hidden widths, layer families, initialization and preprocessing. They must retain declared input/output names and widths, prediction leads, feedback routes/delays, and resampling settings. Returned settings must be finite JSON-compatible Python data; tuples are retained during reconstruction. Capture modules, initializer objects and callables in the factory instead of returning them as trial parameters. Forward the supplied seed when constructing reservoirs.

    Scoring¶

    Argument Default Contract
    threshold 0.2 Positive threshold for the RMS of per-feature normalized errors. Accuracy requires error strictly below this value.
    aggregation "weakest" Minimum head horizon; "mean" averages head horizons.
    objective None objective(ScoreContext) -> finite scalar replaces the ranking.
    pruning_objective None Optional pruning_objective(PrefixScoreContext) -> finite scalar enables custom within-forecast pruning while preserving the complete final objective.
    direction "maximize" Built-in horizon uses maximize. Set "minimize" for a custom error objective.

    The default counts each batch item's contiguous accurate prefix, stopping at the first threshold failure. Each head uses the batch median, averaging the two middle values for an even batch. "weakest" selects the shortest head horizon. Thus horizons of 400 and 100 yield 100. The trial score is the median across its initialization-seed/origin evaluations. Scores are positive sample counts; zero is possible. The winner is refitted with the first seed in the bank.

    Scoring divides each feature's original-unit error by a fixed training-label population standard deviation. The scale uses the prototype's absolute washout, not a trial's tuned washout or any validation prefix. Constant features use scale one; nonconstant scales have a dtype-epsilon floor. Half-precision scale calculation uses float32. This scoring scale is separate from model preprocessing.

    ScoreContext fields are named predictions, truth, scales, plus seed and origin. Predictions/truth retain the data's 2-D/3-D layout and original units. Truth and scale tensors are cloned for custom objectives. A callback receives one full seed/origin forecast and returns one finite scalar; the tuner takes its median across runs. Built-in diagnostics remain available with a custom objective. Nonfinite forecasts, scales or objective values fail the evaluation.

    PrefixScoreContext adds completed_steps and the complete requested horizon. Its prediction/truth tensors contain only the completed prefix. A prefix callback controls pruning, not final ranking, and must return comparable finite values at matching resource steps with the study's direction. Without this explicit callback, a custom complete objective is evaluated only on full forecasts; pruning can then skip later origins/seeds but not shorten the current forecast.

    nrmse is a diagnostic: each head's full-window RMS normalized error, averaged across heads with feature-count weights, then aggregated by the median across runs. It does not break ties in the default horizon objective. Trajectory reports include dt; multiply a horizon by it for physical duration.

    Execution and budgets¶

    Argument Default Contract
    n_jobs 1 Caller process by default; other nonzero integers use joblib workers (-1 selects available CPUs).
    inner_max_num_threads 1 Positive worker BLAS-thread limit.
    timeout None Stops dispatching new windows after elapsed seconds; preparation counts toward this time. Running windows and final refitting finish.
    catch_exceptions True Failed candidate evaluations become Optuna FAIL trials; False raises the original exception. Setup/search errors still raise.
    execution "auto" Select fast step execution when supported, history replay for temporal DAGs; "fast" and "history" force a mode.
    pruner None Registry key or configured Optuna pruner; the parent owns pruning decisions. A supplied study owns its pruner instead.
    checkpoint_interval 64 Positive scored-sample chunk length. Rejection at a checkpoint prevents later forecast chunks/evaluations.
    screening False True uses Screening(); a policy customizes cheap repetition counts, promotion fraction and cohort size. Full evaluation remains the default.
    memory_budget None Integer bytes, a size string or MemoryBudget; estimated admission/concurrency policy, never a process/RSS/GPU allocator cap.
    memory_estimator None estimator(params, MemoryContext) -> MemoryEstimate or int; required with a custom factory and budget. Returns complete totals, floored by known resident requirements.

    The parent samples a finite candidate window before workers evaluate it. Failed candidates consume attempts, and no window dispatches beyond the remaining trial budget. A timeout is a dispatch boundary, not cancellation of an ongoing fit/forecast and not a hard total deadline. Preparation and bookkeeping also consume elapsed time. A run with no valid completed candidate raises.

    Workers receive prepared numerical jobs, without a study or storage connection. The parent owns sampling, real Optuna trial reports and pruning. Joblib/loky owns process creation, cancellation and reuse; there is no fork-based worker launcher. Errors/interruptions cancel outstanding work and finalize this call's unfinished reservations without replacing the original model.

    All candidates use the same fixed data/seed protocol, but an adaptive sampler can propose a different sequence at a different worker count. Caller Python, NumPy and Torch RNG states are restored around search/evaluation callbacks. Fit/forecast timings synchronize accelerators before recording elapsed time. Fit time is recorded once per seed, on its first origin; later origins record zero fit time. Outer module identity, mode and hooks survive successful tuning; parameters can be replaced. Create optimizers afterward. A failed final refit or owned-storage cleanup leaves the original model unchanged.

    Built-in scoring accumulates fixed-scale statistics while forecasts stream. Resource steps are completed_seed_origin_count * horizon + completed_prefix, so intermediate values from matching resources are comparable across trials. The reported intermediate value is the median of completed evaluations and the current prefix score. No pruner is used during the optional screening stage.

    The common pruner="threshold" preset uses ThresholdPruner(lower=0.1, n_warmup_steps=2 * checkpoint_interval) for positive horizon scores. It rejects that string preset with a custom objective; pass a configured Optuna instance with appropriate bounds instead. Other shared keys are "none", "median", "asha" and "hyperband". The sample-accuracy threshold=0.2 is a separate setting.

    Screening uses the same complete horizon with fewer seeds/origins, then ranks each cohort and fully evaluates the promoted configurations. With balanced defaults, all 24 candidates use one seed/origin first; the best eight receive all three seeds and three origins. Only full evaluations become eligible COMPLETE winners. Nonpromoted, oversized and timeout-unsubmitted candidates are recorded as PRUNED, with reasons. A later full evaluation can still be pruned by Optuna. See tuning resources.

    Memory admission accounts for complete winner refits, known long warmup traces, retained custom-objective predictions and history replay. A custom factory changing execution capabilities must estimate its own temporary/replay needs. Individually oversized candidates are skipped before construction; estimated concurrency is reduced for candidates that fit individually. The source model's architecture and precision are never silently changed to satisfy a budget.

    Optuna studies and resumption¶

    Argument Default Contract
    study None Caller-owned single-objective Optuna study with a matching direction and protocol. Queued trials are supported.
    sampler seeded TPESampler Used when ResDAG creates the study.
    storage memory Journal path, SQLite .db path or supported storage URL.
    study_name generated Persistent study name; provide it with storage to resume a named study.
    warm_start None Source Optuna study whose complete configurations are queued for reevaluation by default.
    transfer_mode "reevaluate" "scores" explicitly imports evaluated values only for identical bound protocols and objective directions.

    Pass study= or storage/study-name/sampler/pruner settings. A supplied study owns its sampler, pruner and connections. An unbound study may contain queued WAITING configurations; unverified evaluated scores cannot enter the new protocol.

    The protocol binds prepared data, splits/origins/scales, task timing, seed bank, model configuration, search/objective/recipe configuration and relevant library environment. Changed data, callables or model configuration require a fresh study. Keep callbacks deterministic from their arguments and captured settings. The common tuner checks protocol compatibility before appending attempts. Resuming the same fitted model reuses its original candidate template, rather than treating the fitted winner as a new starting experiment.

    Settings transfer preserves fixed returned values, suggestions and source study/trial provenance. Reevaluation queues single-objective results best-first (trial number breaks ties). Queued configurations consume attempts when run; those beyond this call's budget remain WAITING. Imported scores do not consume new attempts, but only exact matching protocols/directions allow them to compete.

    Owned RDB sessions/pools are released before installing the winner. Returned studies remain usable and reconnect when required; SQLite memory pools remain alive while needed to retain the study. Caller-owned studies retain ownership.

    Raw execution and pruning¶

    Raw forecasts are generated in chunks. Each growing-prefix report reaches the parent pruner before later chunks are generated. A rejection stops future forecast generation. n_checkpoints=5 requests five checkpoints by default; short horizons can yield fewer distinct checkpoints.

    The raw losses retain their existing prefix-dependent normalization, so raw scoring can retain a complete prediction prefix: O(horizon) storage. Custom and monitor losses receive full-horizon tensors when an evaluation completes. With torch_scoring=True, supported losses stay on the tensor device until their final scalar; custom losses may use the NumPy fallback. Attractor-dimension scoring uses NumPy.

    Registry keys are "none", "median", "asha", "hyperband" and "threshold". Raw defaults are minimization losses; choose a configured pruner with suitable bounds for a custom metric. clip_value/prune_on_clip is a separate raw-loss ceiling and can be combined with a pruner. The common tuner's positive threshold preset is not the raw preset.

    memory_budget accepts the same MemoryBudget, size string or byte count. An arbitrary raw factory requires memory_estimator(params, context) when a budget is set. Known prepared data, transfers and retained output storage form a minimum; oversized candidates are PRUNED before factory construction. Parent data preparation happens before estimation, and a pending configuration may remain resident while an admitted window runs. This is planning, not a hard process/RSS/device limit. See tuning resources.

    inner_max_num_threads=1 limits native threads in numerical worker processes. The parent environment, thread settings, logging and Python/NumPy/Torch RNG streams are preserved/restored. seed initializes the sampler and each trial's seed + trial.number RNG scope. Make the factory accept and forward a seed keyword; an explicit model seed in resolved settings takes precedence. Forwarding it also controls initializers with their own local generators. Adaptive proposals can change with worker count and scheduling; exact reproducibility is not a cross-device or cross-schedule guarantee.

    Reuse and transfer¶

    run_hpo(..., warm_start=[config]) queues explicit settings. run_hpo(..., transfer_from=source) queues complete valid settings from a study or storage path for reevaluation by default. Data-only suggestions and fixed model values are preserved. Single-objective configurations are ordered best first, then by trial number; multi-objective sources retain source order.

    Use export_best_config(study) when fixed returned values matter; study.best_params contains only Optuna suggestions. Incompatible model settings must be adapted explicitly. Source study/trial/protocol provenance is retained under the trial's resdag_transfer attribute.

    transfer_mode="scores" requires exactly matching bound task protocols and objective directions. Raw resumption and score import replay historical data preparation with Optuna FixedTrial and check the actual prepared tensor contents. A file changed at the same path is therefore a changed task. Callbacks can run again during validation; keep preparation free of side effects.

    When historical preparation cannot run without live-study resources, supply task_context="dataset-v3" or portable metadata. This explicitly asserts the identity of opaque resources, so update it when they change. Successful replay still checks tensor contents. New attempts receive a real Optuna trial. Use settings reevaluation for changed experiments rather than copying scores.

    def transfer_trials(
    study: optuna.Study,
    transfer_from: "optuna.Study | str",
    *,
    mode: Literal["reevaluate", "scores"] = "reevaluate",
    protocol_hash: str | None = None,
) -> int:
    """Reuse complete configurations, re-evaluating them by default.

    `mode="reevaluate"` enqueues full recorded resolved settings, including
    fixed values that were not Optuna suggestions. Old scores are not copied.
    Source study/trial/protocol provenance is retained on every queued trial.
    A changed objective direction is allowed because the new task evaluates
    the settings again. No parameter is silently removed to fit a new space.

    are bound to exactly `protocol_hash` and their directions match. The
    caller's protocol must identify actual data, validation, objective and
    construction settings. An unbound or changed task must re-evaluate instead.
    All source parameters/distributions and complete resolved settings remain
    intact; incompatible model settings should be adapted explicitly before
    enqueueing, never combined with an old score.

    Returns the number of queued or copied configurations. Invalid/nonfinite
    source evaluations are skipped. Caller-owned study connections are not
    closed; a source opened from a path is released on success and failure.
    """
    if mode not in ("reevaluate", "scores"):
        raise ValueError("mode must be 'reevaluate' or 'scores'")
    with _source_study(transfer_from) as source:
        _require_distinct_studies(study, source)
        source_protocol = source.user_attrs.get(PROTOCOL_KEY)
        if mode == "scores":
            if (
                not protocol_hash
                or source_protocol != protocol_hash
                or study.user_attrs.get(PROTOCOL_KEY) != protocol_hash
            ):
                raise ValueError(
                    "Copying evaluated scores requires an identical bound task protocol; "
                    "use mode='reevaluate' to reuse settings for a new task"
                )
            if study.directions != source.directions:
                raise ValueError("Identical-score transfer requires matching objective directions")
        transferred = 0

        if mode == "reevaluate" and len(source.directions) == 1:

            source_trials.sort(
                key=lambda trial: (
                    (
                        (-(trial.value or 0.0) if maximize else trial.value or 0.0)
                        if trial.value is not None and isfinite(trial.value)
                        else float("inf")
                    ),
                    trial.number,
                )
            )
        entries = []
        for src in source_trials:
            values = src.values
            if (
                not values
                or not all(isfinite(value) for value in values)
                or "error" in src.user_attrs
                or src.user_attrs.get("nonfinite_loss", False)
            ):
                continue
            settings = trial_settings(src)
            if not settings:
                continue

        # Storage write failures can still interrupt a standard Optuna import.
        for src, settings in entries:
            values = src.values
            provenance = {
                "mode": mode,
                "study_name": source.study_name,
                "trial_number": src.number,
                "protocol_hash": source_protocol,
                "resolved_settings_literal": repr(settings),

                    {**src.params, **settings},
                    user_attrs={"resdag_transfer": provenance, REUSED_SETTINGS_KEY: repr(settings)},

    def estimate_memory(
    params: Mapping[str, Any],
    context: MemoryContext,
    *,
    custom_factory: bool = False,
    estimator: Callable[[dict[str, Any], MemoryContext], MemoryEstimate | int] | None = None,

    """Estimate peak fit/forecast/refit bytes, before candidate construction.

    Automatic planning reserves dense preset weights, retained/staged model
    copies, full-refit feature traces and 8-byte QR/SVD/Gram workspaces. On an
    accelerator it also reserves a CPU solve bridge. This deliberately favors
    overestimation; custom layers and factories should provide their own
    estimator. A custom result is a complete estimate, not an added surcharge;
    it cannot remove known resident dataset/model or retained-prefix storage.
    """
    if not isinstance(params, Mapping) or not isinstance(context, MemoryContext):
        raise TypeError("estimate_memory expects parameter mapping and MemoryContext")
    if estimator is not None:
        if not callable(estimator):
            raise TypeError("memory estimator must be callable")
        result = estimator(copy.deepcopy(dict(params)), context)
        if not isinstance(result, MemoryEstimate):
            result = MemoryEstimate(_integer(result, "custom memory estimate"))
        host = context.dataset_host_bytes + context.model_host_bytes
        device = context.dataset_device_bytes + context.model_device_bytes
        retained = context.retained_forecast_bytes + context.replay_history_bytes
        if context.device.type == "cpu":
            host += retained
        else:
            device += retained
        return MemoryEstimate(
            max(result.host_bytes, host),
            max(result.device_bytes, device),
            {
                **result.details,
                "requested_host_bytes": result.host_bytes,
                "requested_device_bytes": result.device_bytes,
                "minimum_host_bytes": host,
                "minimum_device_bytes": device,
                "retained_forecast_bytes": context.retained_forecast_bytes,
                "replay_history_bytes": context.replay_history_bytes,
                "floor_applied": result.host_bytes < host or result.device_bytes < device,
            },
        )
    if custom_factory:
        raise ValueError(
            "A custom factory with a memory budget requires memory_estimator(params, context)"
        )
    if not context.readout_features:
        raise ValueError("Unknown readout widths require a custom memory estimator")
    element = torch.finfo(context.dtype).bits // 8
    n = max(context.train_samples, context.refit_samples)
    features, outputs = context.readout_features, context.readout_outputs
    model = context.model_host_bytes + context.model_device_bytes
    activation_width = context.activation_width
    if context.preset_options:
        options = {**context.preset_options, **params}
        width = _integer(options.get("reservoir_size", 500), "reservoir_size", minimum=1)
        inputs, output = context.input_features, context.output_features
        skip = options.get("input_to_readout", True)
        if not isinstance(skip, bool):
            raise ValueError("input_to_readout must be a bool")
        feedback = _integer(options.get("feedback_size", inputs), "feedback_size", minimum=1)
        features, outputs = (width + (feedback if skip else 0),), (output,)
        # Dense recurrence, feedback/input projection, vector bias/leak/state
        # and readout; no reservoir, initializer or layer is constructed here.
        model = max(
            model, element * (width * width + width * (inputs + 3) + output * (features[0] + 1))
        )
        activation_width = width * 3 + inputs + output + (feedback if skip else 0)
    resident_host = context.dataset_host_bytes + 3 * context.model_host_bytes
    resident_device = context.dataset_device_bytes + 3 * context.model_device_bytes
    model_copies = 3 * model
    fit_data = 4 * n * (context.input_features + context.output_features) * element
    trace_samples = max(n, context.forecast_samples)
    traces = 4 * trace_samples * activation_width * element
    parallel_components: dict[str, int] = {}
    if context.sequence_layout.get("kind") == "parallel":
        layout = context.sequence_layout
        patches = _integer(layout["patches"], "patches", minimum=1)
        rows = trace_samples * patches
        # Input and target window gathers, projected inputs, retained features
        # and core targets all have patch-batched time axes. Retain the same
        # conservative four-copy allowance as the general trace estimate.

            ("input_windows", "patch_width"),
            ("target_windows", "patch_width"),
            ("projected_inputs", "projection_width"),
            ("features", "feature_width"),
            ("core_targets", "output_width"),

                4 * rows * _integer(layout[width_name], width_name, minimum=1) * element

            4
            * context.batch_size
            * patches
            * _integer(layout["state_width"], "state_width", minimum=1)
            * element
        )
        traces = max(traces, sum(parallel_components.values()))
    forecasts = 3 * max(
        context.retained_forecast_bytes,
        context.batch_size * context.output_features * element,
    )
    # QR augmentation and compact SVD, plus centered/cast inputs, inverse-norm
    # guards and factorization work arrays. Sum heads to cover retained features.
    readout_samples = tuple(
        max(
            n,
            context.readout_train_samples[index] if context.readout_train_samples else n,
            context.readout_refit_samples[index] if context.readout_refit_samples else n,
        )
        for index in range(len(features))
    )
    if context.sequence_layout.get("kind") == "parallel":
        layout = context.sequence_layout
        washout = _integer(params.get("washout", layout["washout"]), "washout")
        pooled = max(
            layout[phase]["window"][0]
            * max(
                0,
                layout[phase]["supervised_stop"] - max(washout, layout[phase]["supervised_start"]),
            )
            for phase in ("train", "refit")
        )
        readout_samples = (max(*readout_samples, pooled),)
    workspace = sum(
        8 * (6 * rows * (f + out) + 12 * f * f + 6 * min(rows, f) ** 2 + 4 * f * out)
        for rows, f, out in zip(readout_samples, features, outputs)
    )
    working = (
        model_copies + fit_data + traces + forecasts + context.replay_history_bytes + workspace
    )
    host, device = resident_host, resident_device
    bridge = 0
    if context.device.type == "cpu":
        host += working
    else:
        device += working
        bridge = workspace
        host += bridge
    return MemoryEstimate(
        host,

            "samples": n,
            "element_bytes": element,
            "model_copies": model_copies,
            "fit_data": fit_data,
            "feature_traces": traces,
            "trace_samples": trace_samples,
            "readout_samples": readout_samples,
            "readout_features": features,
            "parallel_trace_components": parallel_components,
            "forecast_outputs": forecasts,
            "replay_history": context.replay_history_bytes,
            "solve_workspace": workspace,
            "cpu_bridge": bridge,
        },
    )

    Raw losses¶

    losses ¶

    Loss functions for hyperparameter optimization.

    This module provides specialized loss functions for evaluating multi-step forecasts in reservoir computing applications. All functions operate on batched predictions with shape (B, T, D) where B is batch size, T is time steps, and D is the number of dimensions.

    Available Losses
    • "efh" : Expected Forecast Horizon (default, recommended for chaotic systems)
    • "forecast_horizon" : Forecast Horizon Loss (contiguous valid steps)
    • "lyapunov" : Lyapunov-weighted Loss (exponential decay for chaotic systems)
    • "standard" : Standard Loss (mean geometric mean error)
    • "soft_horizon" : Soft valid horizon (cumulative survival probability)
    • "attractor_dimension" : Attractor-fidelity — correlation-dimension mismatch of the forecast vs the truth (climate reconstruction, not short-horizon error)
    Scale conventions

    The threshold-based horizon losses (expected_forecast_horizon, forecast_horizon, soft_valid_horizon) compare a per-timestep error against a fixed threshold. For that comparison to be meaningful the error metric must be scale-free, otherwise the same threshold collapses the horizon to 0 or T depending on the absolute scale of the data. All of these losses therefore default to metric="nrmse" (RMSE normalised by the per-dimension standard deviation of y_true). Selecting a raw-scale metric ("rmse", "mse", "mae") for a threshold-based loss emits a UserWarning, because the threshold is then only valid on unit-scale data.

    Batch aggregation

    Per-timestep errors of shape (B, T) are collapsed across the batch axis with the median in every loss (see _aggregate_batch). The median is robust to outlier trajectories and — unlike the geometric mean — does not collapse to 0 when a single batch element happens to be perfect at a step. When a geometric mean is requested explicitly it is computed on clipped errors so it is zero-safe.

    Example

    from resdag.hpo import LOSSES, get_loss loss_fn = get_loss("efh") loss = loss_fn(y_true, y_pred, threshold=0.2)

    See Also

    resdag.hpo.objective : Objective builder that wraps these losses for Optuna. resdag.hpo.run : High-level HPO orchestrator.

        raise ValueError(f"Shape mismatch: {y_true.shape} vs {y_pred.shape}")

    def _flatten(arr: NDArray[np.floating]) -> NDArray[np.floating]:
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim == 3:
            a = a.reshape(-1, a.shape[-1])  # (B*T, D)
        return a

    d2_true = correlation_dimension(
        _flatten(y_true), n_bins=n_bins, fit_window=fit_window, max_points=max_points, seed=seed
    )
    d2_pred = correlation_dimension(
        _flatten(y_pred), n_bins=n_bins, fit_window=fit_window, max_points=max_points, seed=seed
    )
    return abs(float(d2_pred) - float(d2_true))

    """Generate a human-readable summary of an Optuna study.

    Creates a formatted text summary including study statistics, best trial
    information, and top N trials ranked in the study's objective direction.

    Parameters
    ----------
    study : optuna.Study
        The Optuna study to summarize.
    top_n : int, default=5
        Number of top-performing trials to include.

    Returns
    -------
    str
        Formatted multi-line summary string.

    Example
    -------
    >>> study = optuna.create_study()
    >>> # ... run optimization ...
    >>> print(get_study_summary(study))
    """
    lines = []
    lines.append("=" * 60)
    lines.append("Study Summary")
    lines.append("=" * 60)
    lines.append(f"Study Name: {study.study_name}")

    total_trials = len(study.trials)
    completed = len([t for t in study.trials if t.state == optuna.trial.TrialState.COMPLETE])
    pruned = len([t for t in study.trials if t.state == optuna.trial.TrialState.PRUNED])
    failed = len([t for t in study.trials if t.state == optuna.trial.TrialState.FAIL])

    lines.append(f"Total Trials: {total_trials}")
    lines.append(f"  Completed: {completed}")
    lines.append(f"  Pruned: {pruned}")
    lines.append(f"  Failed: {failed}")
    lines.append("")

    if completed > 0:
        lines.append("-" * 60)
        lines.append("Best Trial")
        lines.append("-" * 60)
        lines.append(f"Trial Number: {study.best_trial.number}")
        lines.append(f"Value: {study.best_value:.6f}")
        lines.append("")
        lines.append("Parameters:")
        for k, v in study.best_params.items():
            if isinstance(v, float):
                lines.append(f"  {k}: {v:.6f}")
            else:
                lines.append(f"  {k}: {v}")
        lines.append("")

        # COMPLETE single-objective trials have a value. Keep a defensive
        # missing-value fallback last in either objective direction.
        completed_trials = [t for t in study.trials if t.state == optuna.trial.TrialState.COMPLETE]
        maximizing = study.direction == optuna.study.StudyDirection.MAXIMIZE
        missing_value = float("-inf") if maximizing else float("inf")
        sorted_trials = sorted(
            completed_trials,
            key=lambda t: t.value if t.value is not None else missing_value,
            reverse=maximizing,
        )

        lines.append("-" * 60)
        lines.append(f"Top {min(top_n, len(sorted_trials))} Trials")
        lines.append("-" * 60)
        for i, trial in enumerate(sorted_trials[:top_n], 1):
            lines.append(f"{i}. Trial {trial.number}: {trial.value:.6f}")

    lines.append("=" * 60)
    return "\n".join(lines)

    def make_study_name(model_creator: Callable) -> str:
    """Generate a stable study name from the model creator callable.

    Creates a study name based on the callable's source file and name. It is
    robust to callables that lack the usual introspection metadata:

    - `functools.partial` objects are unwrapped to their underlying function so
      the wrapped factory's name and source file are used (a bare `partial`
      raises `TypeError` in `inspect.getsourcefile`).
    - Callables without a usable `__name__` (e.g. lambdas or callable class
      instances) fall back to their type name plus a short stable hash of their
      `repr`, so two logically-distinct creators do not collide onto the same
      study name and silently share persisted storage.

    Parameters
    ----------
    model_creator : Callable
        The model creator callable to generate a name from. May be a plain
        function, a `functools.partial`, a lambda, or any callable object.

    Returns
    -------
    str
        Study name in the format `"filename:identifier"`. `filename` is
        `"<unknown>"` when the source file cannot be determined.

    Example
    -------
    >>> def my_model_creator(units):
    ...     return model
    >>> make_study_name(my_model_creator)
    'script:my_model_creator'
    """
    fn = _unwrap_callable(model_creator)

    try:
        src = inspect.getsourcefile(fn) or "<interactive>"
    except (TypeError, OSError):
        src = "<unknown>"

    name = getattr(fn, "__name__", None)
    if not name or name == "<lambda>":
        digest = hashlib.sha1(repr(model_creator).encode("utf-8")).hexdigest()[:8]
        name = f"{type(fn).__name__}-{digest}"

    return f"{Path(src).stem}:{name}"
